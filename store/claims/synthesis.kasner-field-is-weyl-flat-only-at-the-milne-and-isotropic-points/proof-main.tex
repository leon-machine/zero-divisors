with $e_{0}, \dots, e_{3}$ the constant orthonormal physical frame and $B_{i} := e_{0}\wedge e_{i}$; so $\hb(e_{0}) = e_{t}$, $\hb(e_{i}) = t^{-p_{i}}e_{x^{i}}$.

\emph{(i)} \emph{The rotation-gauge field.} The torsion equation $D_{e_\mu}\hb^{-1}(e_\nu) = D_{e_\nu}\hb^{-1}(e_\mu)$ has the unique solution $\Omega(e_{t}) = 0$, $\Omega(e_{x^{i}}) = -p_{i}t^{p_{i}-1}B_{i}$, that is $\omega(e_{0}) = 0$ and $\omega(e_{i}) = \Omega(\hb(e_{i})) = -\tfrac{p_{i}}{t}B_{i}$ (dictionary C10): indeed $\partial_{t}(t^{p_{i}}e_{i}) = p_{i}t^{p_{i}-1}e_{i}$ equals $\Omega(e_{x^{i}})\cdot e_{0} = -p_{i}t^{p_{i}-1}B_{i}\cdot e_{0} = p_{i}t^{p_{i}-1}e_{i}$, and for $i \neq j$, $\Omega(e_{x^{i}})\cdot(t^{p_{j}}e_{j}) = 0$. \emph{The field strength.} With $B_{i}\times B_{j} = -e_{i}\wedge e_{j}$ for $i \neq j$,
\[
R(e_{t}\wedge e_{x^{i}}) = \partial_{t}\Omega(e_{x^{i}}) = -p_{i}(p_{i} - 1)t^{p_{i}-2}B_{i}, \qquad
R(e_{x^{i}}\wedge e_{x^{j}}) = \Omega(e_{x^{i}})\times\Omega(e_{x^{j}}) = -p_{i}p_{j}t^{p_{i}+p_{j}-2}\,e_{i}\wedge e_{j},
\]
so the store-sign curvature operator $\mathcal{R}(B) = -R(\hb(B))$ is diagonal on the frame planes,
\[
\mathcal{R}(B_{i}) = p_{i}(p_{i} - 1)\,t^{-2}B_{i}, \qquad \mathcal{R}(e_{i}\wedge e_{j}) = p_{i}p_{j}\,t^{-2}\,e_{i}\wedge e_{j}.
\]
\emph{Ricci and scalar.} $\mathcal{R}(b) = \partial_{a}\cdot\mathcal{R}(a \wedge b)$ with $e^{i}\cdot B_{i} = -e_{0}$, $e^{0}\cdot B_{j} = e_{j}$ and $e^{i}\cdot(e_{i}\wedge e_{j}) = e_{j}$ gives $\mathcal{R}(e_{0}) = \sum_{i}p_{i}(p_{i} - 1)\,t^{-2}e_{0} = (S - 1)\,t^{-2}e_{0}$ and $\mathcal{R}(e_{j}) = p_{j}\bigl(p_{j} - 1 + \sum_{i \neq j}p_{i}\bigr)t^{-2}e_{j} = 0$ by $\sum_{i}p_{i} = 1$; the scalar is $\mathcal{R} = (S - 1)\,t^{-2}$, and $\mathcal{G}(e_{0}) = \tfrac12(S - 1)t^{-2}e_{0}$, $\mathcal{G}(e_{j}) = -\tfrac12(S - 1)t^{-2}e_{j}$, so under the field equation $\mathcal{G}(a) = -8\pi T(a)$ the source has $T(e_0) = \rho\,e_0$, $T(e_j) = -P e_j$ with $P = \rho = (1 - S)/(16\pi t^{2})$, nonnegative exactly when $S \le 1$. \emph{The Weyl map.} $\mathcal{W}(a \wedge b) = \mathcal{R}(a \wedge b) - \tfrac12(\mathcal{R}(a)\wedge b + a \wedge \mathcal{R}(b)) + \tfrac16\mathcal{R}\,a \wedge b$ (Fact \claimref{synthesis.imported-gtg-dictionary}(v)) gives
\[
\mathcal{W}(B_{i}) = \bigl(p_{i}^{2} - p_{i} - \tfrac12(S - 1) + \tfrac16(S - 1)\bigr)t^{-2}B_{i} = \lambda_{i}B_{i}, \qquad \lambda_{i} = \bigl(p_{i}^{2} - p_{i} + \tfrac{1 - S}{3}\bigr)t^{-2},
\]
and, for $\{i, j, k\} = \{1, 2, 3\}$, $\mathcal{W}(e_{j}\wedge e_{k}) = (p_{j}p_{k} + \tfrac{S - 1}{6})t^{-2}\,e_{j}\wedge e_{k}$, where $p_{j}p_{k} = \tfrac12\bigl((1 - p_{i})^{2} - (S - p_{i}^{2})\bigr)$ makes the coefficient $\lambda_{i}$ again. Since $e_{j}\wedge e_{k} = \pm I_{4}B_{i}$, the Weyl map has $B_{i}$ and $I_{4}B_{i}$ as eigenvectors with the same eigenvalue $\lambda_{i}$ (it is purely electric), as $\mathcal{W}(IB) = I\mathcal{W}(B)$ requires. With $p_{3} = 1 - p_{1} - p_{2}$, $1 - S = 2p_{1} + 2p_{2} - 2p_{1}^{2} - 2p_{2}^{2} - 2p_{1}p_{2}$, and $3t^{2}\lambda_{1} = p_{1}^{2} - 2p_{1}p_{2} - p_{1} - 2p_{2}^{2} + 2p_{2} = E_{1}$, $3t^{2}\lambda_{2} = -E_{2}$; $\sum_{i}\lambda_{i} = (S - 1 + 1 - S)t^{-2} = 0$. So $\mathcal{W} = 0$ exactly when every $\lambda_{i}$ vanishes, that is (by the trace) when $E_{1} = E_{2} = 0$, and otherwise a nonzero eigenvalue is a constant times $t^{-2}$, nonzero on all of $t > 0$. \emph{(ii)} By (i), $\lambda_{1} = \lambda_{2} = \lambda_{3} = 0$ exactly when $p_{i}^{2} - p_{i}$ takes the same value for all $i$, and a value $c$ of $p^{2} - p$ is taken at the two roots $r$ and $1 - r$.
